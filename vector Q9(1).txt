#include <iostream>
#include <vector>
using namespace std;

int main() {
    vector<int> v = {10, 15, 20, 25, 30, 35};

    vector<int> result;
    for (int x : v) {
        if (x % 2 != 0) {
            result.push_back(x);
        }
    }

    cout << "After removing even numbers: ";
    for (int x : result)
        cout << x << " ";

    return 0;
}
