#include <iostream>
using namespace std;

const int rows = 10;
const int col = 10;


void initialseats(int seats[rows][ col])
{
    for (int i = 0; i < rows; i++)
        for (int j = 0; j <  col; j++)
            seats[i][j] = 0;
}


void displaySeats(int seats[rows][col])
{
    cout << "\nBus Seat Layout (0 = Empty, 1 = Reserved)\n\n";
    for (int i = 0; i < rows; i++)
    {
        cout << "Row " << i + 1 << ": ";
        for (int j = 0; j <  col; j++)
        {
            cout << seats[i][j] << " ";
        }
        cout << endl;
    }
}


void reserveSeat(int seats[rows][col])
{
    int row, col;
    cout << "Enter row (1-10): ";
    cin >> row;
    cout << "Enter seat number (1-10): ";
    cin >> col;

    if (seats[row - 1][col - 1] == 0)
    {
        seats[row - 1][col - 1] = 1;
        cout << "Seat reserved successfully.\n";
    }
    else
    {
        cout << "Seat already reserved.\n";
    }
}


void cancelSeat(int seats[rows][col])
{
    int row, col;
    cout << "Enter row (1-10): ";
    cin >> row;
    cout << "Enter seat number (1-10): ";
    cin >> col;

    if (seats[row - 1][col - 1] == 1)
    {
        seats[row - 1][col - 1] = 0;
        cout << "Seat reservation cancelled.\n";
    }
    else
    {
        cout << "Seat is already empty.\n";
    }
}


void countSeatStatus(int seats[rows][col])
{
    int full = 0, partial = 0, empty = 0;

    for (int i = 0; i < rows; i++)
    {
        int reservedCount = 0;

        for (int j = 0; j < col; j++)
        {
            if (seats[i][j] == 1)
                reservedCount++;
        }

        if (reservedCount == 0)
            empty++;
        else if (reservedCount == col)
            full++;
        else
            partial++;
    }

    cout << "\nRow Occupancy Status:\n";
    cout << "Fully Occupied Rows   : " << full << endl;
    cout << "Partially Occupied Rows: " << partial << endl;
    cout << "Empty Rows            : " << empty << endl;
}


int main()
{
    int seats[rows][col];
    int choice;

    initialseats(seats);

    do
    {
        cout << "\n--- Bus Seat Reservation System ---\n";
        cout << "1. Display Seats\n";
        cout << "2. Reserve a Seat\n";
        cout << "3. Cancel a Seat\n";
        cout << "4. Count Seat Status\n";
        cout << "5. Exit\n";
        cout << "Enter your choice: ";
        cin >> choice;

        switch (choice)
        {
        case 1:
            displaySeats(seats);
            break;
        case 2:
            reserveSeat(seats);
            break;
        case 3:
            cancelSeat(seats);
            break;
        case 4:
            countSeatStatus(seats);
            break;
        case 5:
            cout << "Exiting program...\n";
            break;
        default:
            cout << "Invalid choice! Try again.\n";
        }

    } while (choice != 5);

    return 0;
}
